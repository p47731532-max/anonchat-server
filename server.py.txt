import os, asyncio, json, websockets

PORT = int(os.environ.get("PORT", 8765))
TOPICS = {}

async def handler(ws):
    subs = set()
    try:
        async for raw in ws:
            try:
                data = json.loads(raw)
                topic = data.get("topic")
                if not topic or len(topic) != 64: continue
                if data.get("action") == "subscribe":
                    TOPICS.setdefault(topic, set()).add(ws)
                    subs.add(topic)
                elif data.get("action") == "publish":
                    for p in list(TOPICS.get(topic, set())):
                        if p != ws:
                            try: await p.send(raw)
                            except: pass
            except: pass
    finally:
        for t in subs:
            TOPICS.get(t, set()).discard(ws)
            if t in TOPICS and not TOPICS[t]: del TOPICS[t]

async def main():
    async with websockets.serve(handler, "0.0.0.0", PORT):
        print(f"SERVER READY ON PORT {PORT}")
        await asyncio.Future()

if __name__ == "__main__":
    asyncio.run(main())
